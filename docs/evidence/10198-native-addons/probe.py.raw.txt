def guard_python(text):
    start = text.index('stale_native_addons() {')
    start = text.index('  python3 - "$abi" <<\'PY\'\n', start) + len('  python3 - "$abi" <<\'PY\'\n')
    return text[start:text.index('\nPY\n}', start)]
def guard(text, fixture, options=None):
    child = env.copy()
    if options: child['NODE_OPTIONS'] = options
    start = time.monotonic()
    result = subprocess.run(['python3', '-', abi], input=guard_python(text), cwd=fixture, env=child, capture_output=True, text=True, timeout=20)
    return {'exit': result.returncode, 'stale': result.stdout.strip().splitlines(), 'stderr': result.stderr, 'seconds': round(time.monotonic()-start, 3)}
load_script = r"""

"""
