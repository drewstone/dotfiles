def guard_python(text):
    start = text.index('stale_native_addons() {')
    start = text.index('  python3 - "$abi" <<\'PY\'\n', start) + len('  python3 - "$abi" <<\'PY\'\n')
    return text[start:text.index('\nPY\n}', start)]
def guard(text, fixture, options=None):

def live(pid):
    try: return pathlib.Path('/proc',str(pid),'stat').read_text().rsplit(')',1)[1].split()[0] != 'Z'
    except FileNotFoundError: return False
def own_children(pidfile):
    return [int(n) for n in pidfile.read_text().splitlines()] if pidfile.exists() else []
def cleanup_owned(pidfile, nonce):
    for pid in own_children(pidfile):
        try:
            environment=pathlib.Path('/proc',str(pid),'environ').read_bytes().split(b'\0')
            if ('NATIVE_PROOF_NONCE='+nonce).encode() in environment: os.kill(pid,signal.SIGKILL)
        except (FileNotFoundError, ProcessLookupError): pass

def first_scan(text, fixture, pidfile, nonce):
    start=text.index('  reap_leftovers() {');reaper=text[start:text.index('\n  stop() {',start)]
    start=text.index('rebuild_native_addons() {');scan=text[start:text.index('  [ -n "$stale" ] || return 0',start)]
    fn=text[text.index('stale_native_addons() {'):text.index('\n# Rebuild the stale native addons')]
    command='run_token='+nonce+'\nreaped=0\n'+reaper+'\n'+fn+'\n'+scan+'  printf \'%s\\n\' "$stale"\n}\nrebuild_native_addons\n'
    child=env.copy();child['NATIVE_PROOF_NONCE']=nonce
    start=time.monotonic();proc=subprocess.Popen(['bash','-c',command],cwd=fixture,env=child,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True,start_new_session=True)
    timed_out=False
    try: stdout,stderr=proc.communicate(timeout=22)
    except subprocess.TimeoutExpired:
        timed_out=True;os.killpg(proc.pid,signal.SIGKILL);stdout,stderr=proc.communicate(timeout=3)
    row={'exit':proc.returncode,'stale':stdout.strip().splitlines(),'stderr':stderr,'outerDeadlineExceeded':timed_out,'seconds':round(time.monotonic()-start,3),'helperPids':own_children(pidfile),'helpersAlive':[pid for pid in own_children(pidfile) if live(pid)]}
    cleanup_owned(pidfile,nonce)
    return row
